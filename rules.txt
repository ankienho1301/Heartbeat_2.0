
rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // =========================
    // COMMON FUNCTIONS
    // =========================

    function signedIn() {
      return request.auth != null;
    }

    function isMember(pairId) {
      return signedIn()
        && request.auth.uid in
          get(
            /databases/$(database)/documents/pairs/$(pairId)
          ).data.members;
    }

    // =========================
    // USER PROFILES
    // =========================

    match /users/{userId} {
      allow read: if signedIn()
        && request.auth.uid == userId;

      allow create: if signedIn()
        && request.auth.uid == userId
        && request.resource.data.uid == userId
        && request.resource.data.displayName is string
        && request.resource.data.displayName.size() >= 1
        && request.resource.data.displayName.size() <= 30;

      allow update: if signedIn()
        && request.auth.uid == userId
        && request.resource.data.uid == userId
        && request.resource.data.displayName is string
        && request.resource.data.displayName.size() >= 1
        && request.resource.data.displayName.size() <= 30;

      allow delete: if false;
    }

    // =========================
    // PAIRS
    // =========================

    match /pairs/{pairId} {

      allow create: if signedIn()
        && request.resource.data.createdBy == request.auth.uid
        && request.resource.data.members == [request.auth.uid]
        && request.resource.data.status == 'waiting'
        && request.resource.data.inviteCode is string
        && getAfter(
          /databases/$(database)/documents/pairCodes/$(request.resource.data.inviteCode)
        ).data.pairId == pairId
        && getAfter(
          /databases/$(database)/documents/pairCodes/$(request.resource.data.inviteCode)
        ).data.createdBy == request.auth.uid;

      allow read: if signedIn()
        && (
          request.auth.uid in resource.data.members
          || (
            resource.data.status == 'waiting'
            && get(
              /databases/$(database)/documents/pairCodes/$(resource.data.inviteCode)
            ).data.status == 'open'
          )
        );

      allow update: if signedIn()
        && resource.data.status == 'waiting'
        && request.resource.data.status == 'paired'
        && request.resource.data.members.size() == 2
        && request.resource.data.members.hasAll(resource.data.members)
        && request.auth.uid in request.resource.data.members
        && !(request.auth.uid in resource.data.members)
        && getAfter(
          /databases/$(database)/documents/pairCodes/$(resource.data.inviteCode)
        ).data.status == 'used'
        && getAfter(
          /databases/$(database)/documents/pairCodes/$(resource.data.inviteCode)
        ).data.joinedBy == request.auth.uid;

      allow delete: if false;

      // =========================
      // PAIR PROFILES
      // =========================

      match /profiles/{userId} {
        allow read: if isMember(pairId);

        allow create, update: if isMember(pairId)
          && request.auth.uid == userId
          && request.resource.data.displayName is string
          && request.resource.data.displayName.size() >= 1
          && request.resource.data.displayName.size() <= 24;

        allow delete: if false;
      }

      // =========================
      // DEVICES
      // =========================

      match /devices/{userId} {
        allow read: if isMember(pairId)
          && request.auth.uid == userId;

        allow create, update: if isMember(pairId)
          && request.auth.uid == userId
          && request.resource.data.token is string;

        allow delete: if isMember(pairId)
          && request.auth.uid == userId;
      }

      // =========================
      // SIGNALS
      // =========================

      match /signals/{signalId} {
        allow read: if isMember(pairId);

        allow create: if isMember(pairId)
          && request.resource.data.senderUid == request.auth.uid
          && request.resource.data.text is string
          && request.resource.data.isLove is bool;

        allow update, delete: if false;
      }
      match /diaryEntries/{entryId} {
  allow read: if isMember(pairId);

  allow create: if isMember(pairId)
    && request.resource.data.authorUid == request.auth.uid
    && request.resource.data.authorName is string
    && request.resource.data.text is string
    && request.resource.data.text.size() >= 1
    && request.resource.data.text.size() <= 2000;

  allow update, delete: if false;
}
      // =========================
      // DENY OTHER SUBCOLLECTIONS
      // =========================

      match /{document=**} {
        allow read, write: if false;
      }
    }

    // =========================
    // PAIR CODES
    // =========================

    match /pairCodes/{code} {
      allow get: if signedIn();
      allow list: if false;

      allow create: if signedIn()
        && request.resource.data.createdBy == request.auth.uid
        && request.resource.data.status == 'open'
        && request.resource.data.pairId is string
        && getAfter(
          /databases/$(database)/documents/pairs/$(request.resource.data.pairId)
        ).data.createdBy == request.auth.uid
        && getAfter(
          /databases/$(database)/documents/pairs/$(request.resource.data.pairId)
        ).data.inviteCode == code;

      allow update: if signedIn()
        && resource.data.status == 'open'
        && request.resource.data.status == 'used'
        && request.resource.data.joinedBy == request.auth.uid
        && request.resource.data.pairId == resource.data.pairId
        && getAfter(
          /databases/$(database)/documents/pairs/$(resource.data.pairId)
        ).data.status == 'paired'
        && request.auth.uid in getAfter(
          /databases/$(database)/documents/pairs/$(resource.data.pairId)
        ).data.members;

      allow delete: if false;
    }
  }
}
