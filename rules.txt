
rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // =========================
    // COMMON FUNCTIONS
    // =========================

    function signedIn() {
      return request.auth != null;
    }

    function isMember(pairId) {
      return signedIn()
        && request.auth.uid in
          get(
            /databases/$(database)/documents/pairs/$(pairId)
          ).data.members;
    }

    function validDisplayName(data) {
      return data.displayName is string
        && data.displayName.size() >= 1
        && data.displayName.size() <= 30;
    }

    function validGender(data) {
      return data.gender in ['male', 'female'];
    }

    function validPairLink(userId) {
      let pairPath = /databases/$(database)/documents/pairs/$(request.resource.data.pairId);
      let pairData = get(pairPath).data;
      let codePath = /databases/$(database)/documents/pairCodes/$(pairData.inviteCode);
      let codeData = get(codePath).data;

      return request.resource.data.pairId is string
        && (!('inviteCode' in request.resource.data)
          || request.resource.data.inviteCode == pairData.inviteCode)
        && (
          (
            pairData.status == 'waiting'
            && pairData.createdBy == userId
            && pairData.members == [userId]
            && request.resource.data.pairStatus == 'waiting'
            && codeData.status == 'open'
            && codeData.pairId == request.resource.data.pairId
          )
          ||
          (
            pairData.status == 'paired'
            && pairData.members.size() == 2
            && userId in pairData.members
            && request.resource.data.pairStatus == 'paired'
            && request.resource.data.partnerUid in pairData.members
            && request.resource.data.partnerUid != userId
            && request.resource.data.partnerName is string
            && codeData.status == 'used'
            && (
              (pairData.createdBy == userId
                && codeData.joinedBy == request.resource.data.partnerUid)
              || (pairData.createdBy != userId
                && codeData.joinedBy == userId)
            )
            && codeData.pairId == request.resource.data.pairId
          )
        );
    }

    // =========================
    // USER PROFILES
    // =========================

    match /users/{userId} {
      allow read: if signedIn()
        && request.auth.uid == userId;

      allow create: if signedIn()
        && request.auth.uid == userId
        && request.resource.data.uid == userId
        && validDisplayName(request.resource.data)
        && validGender(request.resource.data)
        && request.resource.data.keys().hasOnly([
          'uid', 'displayName', 'gender', 'createdAt', 'updatedAt'
        ]);

      allow update: if signedIn()
        && request.auth.uid == userId
        && request.resource.data.uid == userId
        && validDisplayName(request.resource.data)
        && (!('gender' in request.resource.data)
          || validGender(request.resource.data))
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'displayName', 'gender', 'updatedAt', 'pairId', 'pairStatus', 'inviteCode',
          'partnerUid', 'partnerName'
        ])
        && (
          !('pairId' in resource.data)
          || request.resource.data.pairId == resource.data.pairId
        )
        && (
          !request.resource.data.diff(resource.data).affectedKeys().hasAny([
            'pairId', 'pairStatus', 'inviteCode', 'partnerUid', 'partnerName'
          ])
          || validPairLink(userId)
        );

      allow delete: if false;
    }

    // =========================
    // PAIRS
    // =========================

    match /pairs/{pairId} {

      allow create: if signedIn()
        && request.resource.data.createdBy == request.auth.uid
        && request.resource.data.members == [request.auth.uid]
        && request.resource.data.status == 'waiting'
        && request.resource.data.inviteCode is string
        && getAfter(
          /databases/$(database)/documents/pairCodes/$(request.resource.data.inviteCode)
        ).data.pairId == pairId
        && getAfter(
          /databases/$(database)/documents/pairCodes/$(request.resource.data.inviteCode)
        ).data.createdBy == request.auth.uid;

      allow read: if signedIn()
        && (
          request.auth.uid in resource.data.members
          || (
            resource.data.status == 'waiting'
            && get(
              /databases/$(database)/documents/pairCodes/$(resource.data.inviteCode)
            ).data.status == 'open'
          )
        );

      allow update: if (signedIn()
        && resource.data.status == 'waiting'
        && request.resource.data.status == 'paired'
        && request.resource.data.members.size() == 2
        && request.resource.data.members.hasAll(resource.data.members)
        && request.auth.uid in request.resource.data.members
        && !(request.auth.uid in resource.data.members)
        && getAfter(
          /databases/$(database)/documents/pairCodes/$(resource.data.inviteCode)
        ).data.status == 'used'
        && getAfter(
          /databases/$(database)/documents/pairCodes/$(resource.data.inviteCode)
        ).data.joinedBy == request.auth.uid)
        || (signedIn()
          && request.auth.uid in resource.data.members
          && resource.data.status == 'paired'
          && request.resource.data.members == resource.data.members
          && request.resource.data.status == resource.data.status
          && request.resource.data.diff(resource.data).affectedKeys()
            .hasOnly(['anniversaryDate', 'updatedAt'])
          && request.resource.data.anniversaryDate is string
          && request.resource.data.anniversaryDate.matches('^\\d{4}-\\d{2}-\\d{2}$'));

      allow delete: if false;

      // =========================
      // PAIR PROFILES
      // =========================

      match /profiles/{userId} {
        allow read: if isMember(pairId);

        allow create, update: if isMember(pairId)
          && request.auth.uid == userId
          && request.resource.data.displayName is string
          && request.resource.data.displayName.size() >= 1
          && request.resource.data.displayName.size() <= 30
          && (!('gender' in request.resource.data)
            || validGender(request.resource.data));

        allow delete: if false;
      }

      // =========================
      // DEVICES
      // =========================

      match /devices/{userId} {
        allow read: if isMember(pairId)
          && request.auth.uid == userId;

        allow create, update: if isMember(pairId)
          && request.auth.uid == userId
          && request.resource.data.token is string;

        allow delete: if isMember(pairId)
          && request.auth.uid == userId;
      }

      // =========================
      // SIGNALS
      // =========================

      match /signals/{signalId} {
        allow read: if isMember(pairId);

        allow create: if isMember(pairId)
          && request.resource.data.senderUid == request.auth.uid
          && request.resource.data.text is string
          && request.resource.data.isLove is bool;

        allow update, delete: if false;
      }
      match /diaryEntries/{entryId} {
  allow read: if isMember(pairId);

  allow create: if isMember(pairId)
    && request.resource.data.authorUid == request.auth.uid
    && request.resource.data.authorName is string
    && request.resource.data.text is string
    && request.resource.data.text.size() >= 1
    && request.resource.data.text.size() <= 2000;

  allow update, delete: if false;
}
      // =========================
      // DENY OTHER SUBCOLLECTIONS
      // =========================

      match /{document=**} {
        allow read, write: if false;
      }
    }

    // =========================
    // PAIR CODES
    // =========================

    match /pairCodes/{code} {
      allow get: if signedIn();
      allow list: if false;

      allow create: if signedIn()
        && request.resource.data.createdBy == request.auth.uid
        && request.resource.data.status == 'open'
        && request.resource.data.pairId is string
        && getAfter(
          /databases/$(database)/documents/pairs/$(request.resource.data.pairId)
        ).data.createdBy == request.auth.uid
        && getAfter(
          /databases/$(database)/documents/pairs/$(request.resource.data.pairId)
        ).data.inviteCode == code;

      allow update: if signedIn()
        && resource.data.status == 'open'
        && request.resource.data.status == 'used'
        && request.resource.data.joinedBy == request.auth.uid
        && request.resource.data.pairId == resource.data.pairId
        && getAfter(
          /databases/$(database)/documents/pairs/$(resource.data.pairId)
        ).data.status == 'paired'
        && request.auth.uid in getAfter(
          /databases/$(database)/documents/pairs/$(resource.data.pairId)
        ).data.members;

      allow delete: if false;
    }
  }
}
